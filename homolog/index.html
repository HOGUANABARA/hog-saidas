-- ============================================================ -- 37_precos_para_tratar.sql -- SÓ LEITURA. Não altera nada. -- -- A lista de trabalho de preço, numa consulta só. -- Exporte o CSV direto do Supabase (botão de download do editor). -- -- ------------------------------------------------------------ -- AS TRÊS SITUAÇÕES, e por que cada uma importa -- -- 1 · SEM VALOR — o item tem saída registrada e o cadastro não -- tem preço. Toda saída dele entra no relatório valendo -- R$ 0,00: o custo por médico, por cirurgia e por paciente -- sai menor do que é. São os "lançamentos sem valor de -- compra" que o Painel avisa. -- -- 2 · ACIMA DO MAIOR PAGO — o preço do cadastro está acima de -- tudo que já foi comprado de verdade. Infla custo e faz a -- sugestão de compra parecer mais cara do que é. -- -- 3 · ABAIXO DO MENOR PAGO — é a regra M4 dela, invertida: -- "a média nunca pode ser menor do que o valor mais barato -- de compra". Se está abaixo, o custo está subestimado. -- -- A comparação é sempre contra COMPRA REAL (`nf_itens` com valor -- maior que zero). Amostra grátis fica de fora do cálculo — foi a -- decisão M10: amostra conta em quantidade, não em preço. -- -- Tolerância de 1% para não acusar diferença de centavo. -- ============================================================ with compras as ( select ni.material_id, min(ni.valor_unitario) as menor_pago, max(ni.valor_unitario) as maior_pago, sum(ni.qtd * ni.valor_unitario) / nullif(sum(ni.qtd), 0) as media_ponderada, count(*) as n_compras, max(nf.data_emissao) as ultima_compra from nf_itens ni join notas_fiscais nf on nf.id = ni.nota_fiscal_id where ni.valor_unitario > 0 -- amostra e bonificação fora (M10) group by ni.material_id ), uso as ( select m.id, (select count(*) from movimentacoes mv where mv.material_id = m.id and mv.tipo::text like 'SAIDA%') as saidas_app, (select coalesce(sum(s.qtd), 0) from staging_saidas_hist s where s.nome = m.nome) as un_planilha, (select avg(s.valor_unit) from staging_saidas_hist s where s.nome = m.nome and s.valor_unit > 0) as valor_na_planilha from materiais m ) select case when coalesce(m.ultimo_valor_pago, 0) = 0 then '1 · SEM VALOR' when c.maior_pago is not null and m.ultimo_valor_pago > c.maior_pago * 1.01 then '2 · ACIMA DO MAIOR PAGO' when c.menor_pago is not null and m.ultimo_valor_pago < c.menor_pago * 0.99 then '3 · ABAIXO DO MENOR PAGO' end as situacao, m.id, m.nome, m.marca, m.unidade_medida, m.qtd_kit, m.ultimo_valor_pago as preco_no_cadastro, c.menor_pago, c.maior_pago, round(c.media_ponderada, 2) as media_ponderada, c.n_compras, c.ultima_compra, round(u.valor_na_planilha, 2) as valor_na_planilha, u.saidas_app, u.un_planilha, coalesce((select e.saldo from estoque e where e.material_id = m.id and e.local_id = (select l.id from locais l where l.nome = 'ESTOQUE GERAL' limit 1)), 0) as saldo, case when coalesce(m.ultimo_valor_pago,0) = 0 and c.media_ponderada is not null then 'usar a média ponderada das compras: ' || round(c.media_ponderada,2) when coalesce(m.ultimo_valor_pago,0) = 0 and u.valor_na_planilha is not null then 'sem compra no app; a planilha usava: ' || round(u.valor_na_planilha,2) when coalesce(m.ultimo_valor_pago,0) = 0 then 'nunca teve compra com valor — só amostra ou carga inicial' else 'conferir a nota mais recente' end as sugestao from materiais m left join compras c on c.material_id = m.id left join uso u on u.id = m.id where m.ativo is true -- só o que de fato consome: item sem saída nenhuma não distorce custo and (u.saidas_app > 0 or u.un_planilha > 0) and ( coalesce(m.ultimo_valor_pago, 0) = 0 or (c.maior_pago is not null and m.ultimo_valor_pago > c.maior_pago * 1.01) or (c.menor_pago is not null and m.ultimo_valor_pago < c.menor_pago * 0.99) ) order by 1, coalesce((select e.saldo from estoque e where e.material_id = m.id and e.local_id = (select l.id from locais l where l.nome = 'ESTOQUE GERAL' limit 1)), 0) desc; -- ============================================================ -- BLOCO 2 · CAIXA × UNIDADE -- -- Observação dela em 08/08: "tem itens que o valor unitário é -- compatível com o valor da caixa." -- -- O sistema NÃO tem como saber sozinho qual dos dois é: os dois -- são números válidos. O que ele pode fazer é mostrar os dois -- lado a lado e deixar você decidir em uma passada. -- -- Por que isso importa mais que os outros três casos: um preço -- ausente aparece como R$ 0,00 e salta aos olhos. Um preço de -- caixa gravado como unitário sai plausível e errado por um fator -- igual ao `qtd_kit` — 12, 20, 100. Ele não chama atenção em -- lugar nenhum, e contamina custo por cirurgia, custo por -- paciente e sugestão de compra ao mesmo tempo. -- -- `preco_se_for_caixa` é quanto ficaria a unidade se o valor -- gravado for mesmo o da caixa. Compare com o que você paga. -- ============================================================ select m.id, m.nome, m.marca, m.unidade_medida, m.qtd_kit, m.ultimo_valor_pago as preco_gravado, round(m.ultimo_valor_pago / nullif(m.qtd_kit, 0), 4) as preco_se_for_caixa, (select round(avg(ni.valor_unitario), 2) from nf_itens ni where ni.material_id = m.id and ni.valor_unitario > 0) as media_nas_notas, (select count(*) from nf_itens ni where ni.material_id = m.id and ni.valor_unitario > 0) as n_compras, m.descricao_nf from materiais m where m.ativo is true and coalesce(m.qtd_kit, 1) > 1 and coalesce(m.ultimo_valor_pago, 0) > 0 order by m.ultimo_valor_pago desc; -- ------------------------------------------------------------ -- Resumo em uma linha, para dimensionar antes de abrir o CSV -- ------------------------------------------------------------ -- select situacao, count(*) from ( ...o BLOCO 1... ) x group by 1 order by 1;
